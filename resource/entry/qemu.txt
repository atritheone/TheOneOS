  -append "console=ttyS0 root=/dev/vda rw init=/init" \
  -nographic


quiet

qemu-system-x86_64 \
  -m 512M \
  -kernel T1OSkernel \
  -initrd initramfs.cpio.gz \
  -drive file=storage.img,format=raw,if=virtio \
  -append "console=ttyS0 root=/dev/vda rw init=/init quiet loglevel=0" \
  -nographic \
  -monitor none \
  -serial stdio \
  2>/dev/null
